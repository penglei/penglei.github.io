  { docExt :: Maybe String,
    docTitle :: Text,
    docAuthor :: Maybe Text,
    docTags :: Maybe [Text],
    docDate :: Maybe Text,
    docContent :: Maybe Text,
    docPath :: Maybe Text,
    docAttrs :: Maybe Value
  }
  deriving (Show, Generic)

qsort []     = []
qsort (x:xs) = qsort (filter (< x) xs) ++ [x] ++
